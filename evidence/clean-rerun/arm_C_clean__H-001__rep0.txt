def time_to_meet(a_speed: float, b_speed: float, gap: float, delay: float) -> float:
    relative_speed = a_speed + b_speed
    distance_covered_during_delay = b_speed * delay
    remaining_gap = gap - distance_covered_during_delay
    if remaining_gap <= 0:
        return delay
    time_to_meet = remaining_gap / relative_speed
    return delay + time_to_meet
