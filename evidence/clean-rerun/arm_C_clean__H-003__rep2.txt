def is_geometrically_growing(f: callable, n: int, factor: float) -> bool:
    previous_value = f(1)
    for i in range(2, n + 1):
        current_value = f(i)
        if abs(current_value / previous_value - factor) > 0.01:
            return False
        previous_value = current_value
    return True
