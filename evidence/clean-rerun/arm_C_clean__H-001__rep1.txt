def meet_time(a_speed: float, b_speed: float, gap: float, delay: int) -> str:
    relative_speed = a_speed + b_speed
    time_to_meet_hours = gap / relative_speed
    total_minutes = time_to_meet_hours * 60 + delay
    hours = int(total_minutes // 60)
    minutes = int(total_minutes % 60)
    return f"{hours} hours and {minutes} minutes"

def meet_time(speed1: int, speed2: int, distance: int, unit: int) -> str:
    relative_speed = speed1 + speed2
    time_to_meet = distance / relative_speed
    hours = int(time_to_meet)
    minutes = int((time_to_meet - hours) * 60)
    return f"{hours:02d}:{minutes:02d}"
