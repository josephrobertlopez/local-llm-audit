def calculate_meeting_time(a_speed: float, b_speed: float, gap: float, delay: float) -> float:
    relative_speed = a_speed + b_speed
    effective_gap = gap - a_speed * delay
    meeting_time = delay + effective_gap / relative_speed if effective_gap > 0 else delay
    return meeting_time

def meet_time(duration: int, start_time: datetime, end_time: datetime, buffer: int) -> datetime:
    total_minutes = (end_time - start_time).total_seconds() / 60
    available_minutes = total_minutes - buffer
    if duration > available_minutes:
        raise ValueError("Meeting duration exceeds available time")
    meeting_start = start_time + timedelta(minutes=buffer // 2)
    return meeting_start
