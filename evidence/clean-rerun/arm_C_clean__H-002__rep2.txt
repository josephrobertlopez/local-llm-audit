def select_non_adjacent_books(books: list) -> bool:
    if len(books) <= 1:
        return True
    for i in range(len(books) - 2):
        if abs(books[i] - books[i + 1]) == 1 or abs(books[i + 1] - books[i + 2]) == 1:
            return False
    return True
