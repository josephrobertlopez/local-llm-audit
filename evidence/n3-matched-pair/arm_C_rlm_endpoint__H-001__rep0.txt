def meet_time(a_speed: float, b_speed: float, gap: float, delay: float) -> float:
    total_speed = a_speed + b_speed
    time_to_meet = gap / total_speed
    return time_to_meet + delay

def meet_time(arrival1: str, departure1: str, arrival2: str, departure2: str) -> str:
    from datetime import datetime, timedelta

    def to_minutes(time_str):
        hours, minutes = map(int, time_str.split(':'))
        return hours * 60 + minutes

    def to_time_str(minutes):
        hours, minutes = divmod(minutes, 60)
        return f"{hours:02d}:{minutes:02d}"

    arrival1_minutes = to_minutes(arrival1)
    departure1_minutes = to_minutes(departure1)
    arrival2_minutes = to_minutes(arrival2)
    departure2_minutes = to_minutes(departure2)

    meeting_start = max(arrival1_minutes, arrival2_minutes)
    meeting_end = min(departure1_minutes, departure2_minutes)

    if meeting_start >= meeting_end:
        return "No overlap"

    return f"{to_time_str(meeting_start)} - {to_time_str(meeting_end)}"
