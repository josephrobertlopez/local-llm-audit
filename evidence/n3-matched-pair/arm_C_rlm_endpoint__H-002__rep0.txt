def generate_combinations(books: list) -> list:
    n = len(books)
    combinations = []
    for i in range(n):
        for j in range(i+1, n):
            for k in range(j+1, n):
                for l in range(k+1, n):
                    for m in range(l+1, n):
                        combinations.append([books[i], books[j], books[k], books[l], books[m]])
    return combinations

def filter_adjacent_books(combinations: list) -> list:
    def are_books_adjacent(combination):
        for i in range(len(combination) - 1):
            if abs(combination[i] - combination[i + 1]) == 1:
                return True
        return False

    filtered_combinations = [combination for combination in combinations if not are_books_adjacent(combination)]
    return filtered_combinations
