def time_to_meet(a_speed: float, b_speed: float, gap: float, delay: float) -> float:
    relative_speed = a_speed + b_speed
    effective_gap = gap - a_speed * delay
    return effective_gap / relative_speed if effective_gap > 0 else 0

def meet_time(start1, end1, start2, end2):
