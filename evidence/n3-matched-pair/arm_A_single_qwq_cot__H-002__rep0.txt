def count_non_adjacent(n):
    from itertools import permutations
    # Create a list of n distinct elements, including 'A' and 'B'
    elements = ['A', 'B'] + [chr(ord('C') + i) for i in range(n-2)]
    count = 0
    for p in permutations(elements):
        adjacent = False
        for i in range(len(p)-1):
            if (p[i] == 'A' and p[i+1] == 'B') or (p[i] == 'B' and p[i+1] == 'A'):
                adjacent = True
                break
        if not adjacent:
            count +=1
    return count
