def check_geometric_growth(seq: list) -> bool:
    if len(seq) < 2:
        return False
    ratios = [seq[i] / seq[i - 1] for i in range(1, len(seq))]
    avg_ratio = sum(ratios) / len(ratios)
    for i in range(1, len(seq)):
        if abs(seq[i] / seq[i - 1] - avg_ratio) > 0.1:
            return False
    return True
